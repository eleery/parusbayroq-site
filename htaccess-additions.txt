# Parus Bayroq — строки, которые нужно ДОБАВИТЬ в существующий .htaccess в корне сайта.
# Файл целиком не заменять: в нём могут быть настройки хостинга.

ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
RewriteEngine On
# /index.html и /uz/index.html -> адрес без index.html (убирает дубль главной)
RewriteCond %{THE_REQUEST} \s/+(.*?/)?index\.html[\s?] [NC]
RewriteRule ^(.*?/)?index\.html$ /$1 [R=301,L]
</IfModule>

<IfModule mod_headers.c>
Header always set X-Content-Type-Options "nosniff"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set Strict-Transport-Security "max-age=31536000"
</IfModule>
